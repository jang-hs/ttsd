# Core dependencies for ttsd. Install with `pip install -r requirements.txt`
# (or `uv pip install -r requirements.txt`).
#
# This installs the HTTP server + PyTorch backbone only. Each TTS backend's
# upstream package is installed separately on top of this, because the
# different upstream projects pin mutually-incompatible transformers versions
# (chatterbox-tts -> transformers==5.x, qwen-tts -> transformers==4.57.3, etc.).
# The CLI handles per-backend installs:
#
#   python -m app.cli setup-backends            # all 4 backends
#   python -m app.cli setup-backends kokoro     # one backend
#
# Cross-platform: works on macOS (CPU/MPS), Linux (CPU/CUDA), Windows
# (CPU/CUDA). Device selection is automatic; override with TTSD_DEVICE.

fastapi
uvicorn[standard]
numpy
soundfile

# HF download path
huggingface_hub
hf_transfer==0.1.9

# PyTorch inference backbone. CPU wheels are available on all three OSes from
# pypi. For CUDA acceleration on Linux/Windows, install the matching wheel
# from https://pytorch.org before running this command.
torch>=2.4
torchaudio>=2.4

# Transformers ships HiggsAudioV2ForConditionalGeneration in 4.57+; the
# OmniVoice backend uses it directly without any extra wrapper package.
transformers>=4.57

# Shared tokenizer / utility deps used transitively by the model packages.
sentencepiece
einops

# Only needed by `populate_voices fleurs` (skip if you don't use FLEURS).
datasets

# Test-only dep (still small enough to keep here so `setup` warms the venv
# fully). Remove if you want a leaner runtime install.
pytest
